void sendUIDToForm(const char* uid) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("❌ WiFi not connected, cannot send UID to form");
    return;
  }

  Serial.printf("📤 Sending UID to form: %s\n", uid);

  InsecureWiFiClient client;
  client.setTimeout(2000);
  
  HTTPClient http;
  
  char url[128];
  snprintf(url, sizeof(url), "%s%s/register-uid", serverURL, apiEndpoint.c_str());
  
  http.begin(client, url);
  http.addHeader("Content-Type", "application/json");
  http.setTimeout(2000);
  
  StaticJsonDocument<128> doc;
  doc["uid"] = uid;
  
  String jsonString;
  serializeJson(doc, jsonString);
  
  Serial.print("📦 JSON: ");
  Serial.println(jsonString);
  
  int httpResponseCode = http.POST(jsonString);
  
  Serial.printf("📡 HTTP Response code: %d\n", httpResponseCode);
  
  if (httpResponseCode > 0) {
    String response = http.getString();
    Serial.print("📨 Response: ");
    Serial.println(response);
  }
  
  http.end();
  client.stop();
  feedWatchdog();
}
